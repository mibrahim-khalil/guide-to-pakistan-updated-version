function slugify(text = "") {
  return text
    .toLowerCase()
    .trim()
    .replace(/[’']/g, "")
    .replace(/&/g, "and")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^-|-$/g, "");
}

function riskClass(risk = "") {
  const normalizedRisk = risk.toLowerCase();

  if (normalizedRisk.includes("not safe")) {
    return "notSafe";
  }

  if (normalizedRisk.includes("high")) {
    return "high";
  }

  if (normalizedRisk.includes("moderate")) {
    return "moderate";
  }

  if (normalizedRisk.includes("low")) {
    return "low";
  }

  return "unknown";
}

const RISK_ORDER = {
  "not safe": 1,
  "high risk": 2,
  "moderate risk": 3,
  "low risk": 4,
  unknown: 5,
};

function sortByRisk(places = []) {
  return [...places].sort((firstPlace, secondPlace) => {
    const firstRisk =
      RISK_ORDER[(firstPlace.risk || "unknown").toLowerCase()] || 5;

    const secondRisk =
      RISK_ORDER[(secondPlace.risk || "unknown").toLowerCase()] || 5;

    return firstRisk - secondRisk;
  });
}

export function normalizeSlug(value = "") {
  return String(value)
    .trim()
    .toLowerCase()
    .replace(/&/g, "and")
    .replace(/['’]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

export function toTitleCaseFromSlug(slug = "") {
  return normalizeSlug(slug)
    .split("-")
    .filter(Boolean)
    .map((word) => word.charAt(0).toUpperCase() + word.slice(1))
    .join(" ");
}

export function makePlaceKey(regionSlug = "", placeSlug = "") {
  return `${normalizeSlug(regionSlug)}::${normalizeSlug(placeSlug)}`;
}

export function getRegionLabel(regionSlug = "") {
  const normalizedRegion = normalizeSlug(regionSlug);

  const regionLabels = {
    "gilgit-baltistan": "Gilgit-Baltistan",

    "khyber-pakhtunkhwa": "Khyber Pakhtunkhwa",

    islamabad: "Islamabad Capital Territory",

    punjab: "Punjab",

    sindh: "Sindh",

    balochistan: "Balochistan",

    kashmir: "Azad Jammu & Kashmir",

    "azad-jammu-and-kashmir": "Azad Jammu & Kashmir",
  };

  return (
    regionLabels[normalizedRegion] || toTitleCaseFromSlug(normalizedRegion)
  );
}
