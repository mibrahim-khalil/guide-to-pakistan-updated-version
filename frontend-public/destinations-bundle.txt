import { PLACE_REGISTRY } from "./placeRegistry";

import {
  makePlaceKey,
  resolvePlaceSlug,
  resolveRegionSlug,
} from "../helpers/slugHelpers";

/* =========================================================
   NORMALIZE ONE PLACE
========================================================= */

function normalizePlace(place) {
  if (!place) {
    return null;
  }

  const regionSlug = resolveRegionSlug(
    place.regionSlug,
  );

  const placeSlug = resolvePlaceSlug(
    place.slug,
  );

  if (!regionSlug || !placeSlug) {
    return null;
  }

  return {
    ...place,
    regionSlug,
    slug: placeSlug,
  };
}

/* =========================================================
   NORMALIZED REGISTRY
========================================================= */

const NORMALIZED_PLACE_REGISTRY =
  PLACE_REGISTRY.map(normalizePlace).filter(
    Boolean,
  );

/* =========================================================
   CREATE AUTOMATIC PLACES TO VISIT

   When a place does not have custom placesToVisit data,
   the other destinations from the same region are used.
========================================================= */

function createRegionalPlacesToVisit(
  currentPlace,
) {
  if (!currentPlace) {
    return [];
  }

  const customPlaces =
    currentPlace.placesToVisit;

  if (
    Array.isArray(customPlaces) &&
    customPlaces.length > 0
  ) {
    return customPlaces;
  }

  return NORMALIZED_PLACE_REGISTRY.filter(
    (candidatePlace) =>
      candidatePlace.regionSlug ===
        currentPlace.regionSlug &&
      candidatePlace.slug !==
        currentPlace.slug,
  ).map((candidatePlace) => ({
    id: candidatePlace.slug,

    slug: candidatePlace.slug,

    name: candidatePlace.name,

    region:
      candidatePlace.regionName,

    image:
      candidatePlace.hero?.image,

    description:
      candidatePlace.hero?.summary ||
      `Explore ${candidatePlace.name} and discover its local attractions, culture, and natural surroundings.`,

    path: `/destinations/${candidatePlace.regionSlug}/${candidatePlace.slug}`,
  }));
}

/* =========================================================
   FINAL ENRICHED REGISTRY
========================================================= */

const ENRICHED_PLACE_REGISTRY =
  NORMALIZED_PLACE_REGISTRY.map(
    (place) => ({
      ...place,

      placesToVisit:
        createRegionalPlacesToVisit(place),
    }),
  );

/* =========================================================
   CREATE FAST LOOKUP MAP
========================================================= */

const PLACE_LOOKUP =
  ENRICHED_PLACE_REGISTRY.reduce(
    (lookup, place) => {
      const placeKey = makePlaceKey(
        place.regionSlug,
        place.slug,
      );

      lookup[placeKey] = place;

      return lookup;
    },
    {},
  );

/* =========================================================
   CREATE NESTED PLACE DETAILS OBJECT

   Supports:
   PLACE_DETAILS["punjab"]["murree"]
========================================================= */

export const PLACE_DETAILS =
  ENRICHED_PLACE_REGISTRY.reduce(
    (details, place) => {
      if (!details[place.regionSlug]) {
        details[place.regionSlug] = {};
      }

      details[place.regionSlug][
        place.slug
      ] = place;

      return details;
    },
    {},
  );

/* =========================================================
   GET ONE PLACE
========================================================= */

export function getPlaceDetails(
  regionSlug = "",
  placeSlug = "",
) {
  const resolvedRegionSlug =
    resolveRegionSlug(regionSlug);

  const resolvedPlaceSlug =
    resolvePlaceSlug(placeSlug);

  const placeKey = makePlaceKey(
    resolvedRegionSlug,
    resolvedPlaceSlug,
  );

  return PLACE_LOOKUP[placeKey] || null;
}

/* =========================================================
   GET ALL PLACES
========================================================= */

export function getAllPlaces() {
  return [...ENRICHED_PLACE_REGISTRY];
}

/* =========================================================
   GET PLACES BY REGION
========================================================= */

export function getPlacesByRegion(
  regionSlug = "",
) {
  const resolvedRegionSlug =
    resolveRegionSlug(regionSlug);

  return ENRICHED_PLACE_REGISTRY.filter(
    (place) =>
      place.regionSlug ===
      resolvedRegionSlug,
  );
}

/* =========================================================
   CHECK IF PLACE EXISTS
========================================================= */

export function hasPlaceDetails(
  regionSlug = "",
  placeSlug = "",
) {
  return Boolean(
    getPlaceDetails(
      regionSlug,
      placeSlug,
    ),
  );
}

/* =========================================================
   GET PLACE BY SLUG ONLY
========================================================= */

export function getPlaceBySlug(
  placeSlug = "",
) {
  const resolvedPlaceSlug =
    resolvePlaceSlug(placeSlug);

  return (
    ENRICHED_PLACE_REGISTRY.find(
      (place) =>
        place.slug ===
        resolvedPlaceSlug,
    ) || null
  );
}

/* =========================================================
   GET REGISTERED REGIONS
========================================================= */

export function getRegisteredRegions() {
  return [
    ...new Set(
      ENRICHED_PLACE_REGISTRY.map(
        (place) => place.regionSlug,
      ),
    ),
  ];
}

/* =========================================================
   GET REGISTERED PLACE ROUTES
========================================================= */

export function getRegisteredPlaceRoutes() {
  return ENRICHED_PLACE_REGISTRY.map(
    (place) => ({
      regionSlug:
        place.regionSlug,

      placeSlug:
        place.slug,

      name:
        place.name,

      path: `/destinations/${place.regionSlug}/${place.slug}`,
    }),
  );
}

export function normalizeSlug(value = "") {
  return String(value)
    .trim()
    .toLowerCase()
    .replace(/&/g, "and")
    .replace(/['’]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

export function toTitleCaseFromSlug(slug = "") {
  return normalizeSlug(slug)
    .split("-")
    .filter(Boolean)
    .map((word) => word.charAt(0).toUpperCase() + word.slice(1))
    .join(" ");
}

export function makePlaceKey(regionSlug = "", placeSlug = "") {
  return `${normalizeSlug(regionSlug)}::${normalizeSlug(placeSlug)}`;
}

export function getRegionLabel(regionSlug = "") {
  const normalizedRegion = normalizeSlug(regionSlug);

    "gilgit-baltistan": "Gilgit-Baltistan",

  };

  return (

  );
}

function splitHeading(name) {
  const clean = (name || "")
    .replace(/capital territory/i, "")
    .trim();

  if (clean.toLowerCase().includes("and")) {
    return clean.toUpperCase();
  }

  const parts = clean.split(" ");

  if (parts.length <= 1) {
    return clean.toUpperCase();
  }

  return `${parts[0].toUpperCase()}\n${parts
    .slice(1)
    .join(" ")
    .toUpperCase()}`;
}

function slugify(text = "") {
  return text
    .toLowerCase()
    .trim()
    .replace(/[’']/g, "")
    .replace(/&/g, "and")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^-|-$/g, "");
}

function getRiskClass(risk = "") {
  const normalizedRisk = risk.toLowerCase();

  if (normalizedRisk.includes("not safe")) {
    return "notSafe";
  }

  if (normalizedRisk.includes("high")) {
    return "high";
  }

  if (normalizedRisk.includes("moderate")) {
    return "moderate";
  }

  if (normalizedRisk.includes("low")) {
    return "low";
  }

  return "unknown";
}

const RISK_ORDER = {
  "not safe": 1,
  "high risk": 2,
  "moderate risk": 3,
  "low risk": 4,
  unknown: 5,
};

function sortPlacesByRisk(places = []) {
  return [...places].sort((firstPlace, secondPlace) => {
    const firstRisk =
      RISK_ORDER[(firstPlace.risk || "unknown").toLowerCase()] || 5;

    const secondRisk =
      RISK_ORDER[(secondPlace.risk || "unknown").toLowerCase()] || 5;

    return firstRisk - secondRisk;
  });
}

function chunkIntoGroups(items = [], groupSize = 3) {
  const groups = [];

  for (let index = 0; index < items.length; index += groupSize) {

function slugify(text = "") {
  return text
    .toLowerCase()
    .trim()
    .replace(/[’']/g, "")
    .replace(/&/g, "and")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^-|-$/g, "");
}

function riskClass(risk = "") {
  const normalizedRisk = risk.toLowerCase();

  if (normalizedRisk.includes("not safe")) {
    return "notSafe";
  }

  if (normalizedRisk.includes("high")) {
    return "high";
  }

  if (normalizedRisk.includes("moderate")) {
    return "moderate";
  }

  if (normalizedRisk.includes("low")) {
    return "low";
  }

  return "unknown";
}

const RISK_ORDER = {
  "not safe": 1,
  "high risk": 2,
  "moderate risk": 3,
  "low risk": 4,
  unknown: 5,
};

function sortByRisk(places = []) {
  return [...places].sort((firstPlace, secondPlace) => {
    const firstRisk =
      RISK_ORDER[(firstPlace.risk || "unknown").toLowerCase()] || 5;

    const secondRisk =
      RISK_ORDER[(secondPlace.risk || "unknown").toLowerCase()] || 5;

    return firstRisk - secondRisk;
  });
}
